## Design Guidance
**Domain Entities** (`domain/admin/entities.py`):
```python
from __future__ import annotations
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional

@dataclass
class SystemHealth:
    status: str            # 'healthy' | 'degraded' | 'unhealthy'
    database_connected: bool
    nlp_pipeline_ready: bool
    embedding_model_loaded: bool
    llm_providers_available: list[str]
    uptime_seconds: float
    checked_at: datetime
    issues: list[str] = field(default_factory=list)

@dataclass
class BackgroundTask:
    id: str
    name: str
    status: str            # 'pending' | 'running' | 'completed' | 'failed'
    created_at: datetime
    started_at: Optional[datetime] = None
    completed_at: Optional[datetime] = None
    error: Optional[str] = None
    result: Optional[dict] = None

@dataclass
class AppConfiguration:
    sections: dict         # {section_name: dict_of_values}
    # Wraps config sections as plain dicts.
    # The domain entity does NOT depend on Pydantic.
    # API key values are unmasked here; masking is a presentation concern.
```

**AdminService** (`domain/admin/services.py`):
```python
from datetime import datetime, timezone
from typing import Optional
from domain.admin.entities import SystemHealth, BackgroundTask, AppConfiguration
from domain.admin.ports import MetricsCollector, ConfigurationStore
from domain.admin.exceptions import TaskNotFoundError, ConfigurationError
import uuid

class AdminService:
    def __init__(
        self,
        metrics_collector: MetricsCollector,
        config_store: ConfigurationStore,
    ) -> None:
        self._metrics = metrics_collector
        self._config = config_store
        self._tasks: dict[str, BackgroundTask] = {}  # in-memory

    def check_health(self) -> SystemHealth:
        return self._metrics.collect_health()

    def get_configuration(self) -> AppConfiguration:
        return self._config.load()

    def update_configuration(
        self, section: str, updates: dict
    ) -> AppConfiguration:
        config = self._config.load()
        if section not in config.sections:
            raise ConfigurationError(f'Unknown config section: {section}')
        config.sections[section].update(updates)
        self._config.save(config)
        return config

    def register_task(self, name: str) -> BackgroundTask:
        task = BackgroundTask(
            id=str(uuid.uuid4()), name=name,
            status='pending', created_at=datetime.now(timezone.utc)
        )
        self._tasks[task.id] = task
        return task

    def get_task(self, task_id: str) -> BackgroundTask:
        if task_id not in self._tasks:
            raise TaskNotFoundError(f'Task {task_id} not found')
        return self._tasks[task_id]

    def list_tasks(self) -> list[BackgroundTask]:
        return list(self._tasks.values())

    def update_task_status(
        self, task_id: str, status: str,
        error: Optional[str] = None,
        result: Optional[dict] = None,
    ) -> BackgroundTask:
        task = self.get_task(task_id)
        task.status = status
        if status == 'running':
            task.started_at = datetime.now(timezone.utc)
        elif status in ('completed', 'failed'):
            task.completed_at = datetime.now(timezone.utc)
        task.error = error
        task.result = result
        return task
```
