# 13.2 The Great Normalization - Technical Design

This document details the technical design for normalizing the database structure of `layers`, `domains`, and `terms` into a single `structure_nodes` table, along with related system changes.

## 1. Database Schema Changes

### 1.1 New `structure_nodes` Table

```sql
CREATE TABLE structure_nodes (
    id TEXT PRIMARY KEY NOT NULL DEFAULT (lower(hex(randomblob(16)))),
    node_type TEXT NOT NULL CHECK (node_type IN ('layer', 'domain', 'term')),
    parent_node_id TEXT,
    title TEXT NOT NULL,
    definition TEXT,
    structural_predicate_id TEXT,
    title_embedding BLOB,
    definition_embedding BLOB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    version INTEGER DEFAULT 1,
    last_modified TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (parent_node_id) REFERENCES structure_nodes(id) ON DELETE CASCADE,
    FOREIGN KEY (structural_predicate_id) REFERENCES predicates(id)
);

-- Indexes for performance
CREATE INDEX idx_nodes_node_type ON structure_nodes(node_type);
CREATE INDEX idx_nodes_parent_node_id ON structure_nodes(parent_node_id);
CREATE INDEX idx_nodes_type_parent ON structure_nodes(node_type, parent_node_id);
```

### 1.2 New `structure_node_links` Table

```sql
CREATE TABLE structure_node_links (
    id TEXT PRIMARY KEY NOT NULL DEFAULT (lower(hex(randomblob(16)))),
    source_node_id TEXT NOT NULL,
    target_node_id TEXT NOT NULL,
    predicate TEXT NOT NULL,
    predicate_id TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    FOREIGN KEY (source_node_id) REFERENCES structure_nodes(id) ON DELETE CASCADE,
    FOREIGN KEY (target_node_id) REFERENCES structure_nodes(id) ON DELETE CASCADE,
    FOREIGN KEY (predicate_id) REFERENCES predicates(id),
    
    UNIQUE(source_node_id, target_node_id, predicate)
);

-- Indexes for performance
CREATE INDEX idx_node_links_source ON structure_node_links(source_node_id);
CREATE INDEX idx_node_links_target ON structure_node_links(target_node_id);
```

### 1.3 New `structure_node_events` Table

```sql
CREATE TABLE structure_node_events (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    event_type TEXT NOT NULL,  -- create, update, delete
    node_type TEXT NOT NULL,   -- layer, domain, term, structure_node_link
    old_data TEXT,             -- JSON
    new_data TEXT,             -- JSON
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    processed BOOLEAN DEFAULT FALSE NOT NULL
);

-- Index for performance
CREATE INDEX idx_node_events_processed ON structure_node_events(processed);
```

### 1.4 Vector Embeddings Virtual Table

```sql
-- Virtual table for vector embeddings (sqlite-vec)
CREATE VIRTUAL TABLE structure_nodes_vec USING vec0(
    node_id TEXT PRIMARY KEY,
    title_embedding FLOAT[384],
    definition_embedding FLOAT[384]
);
```

## 2. Data Migration Strategy (006_nodes.py)

### 2.1 Migration Steps

```python
def upgrade():
    """
    Migration 006: Normalize layers, domains, terms into structure_nodes table
    """
    # Step 1: Create new tables
    create_nodes_table()
    create_node_links_table()

    migrate_layers_to_nodes()
    migrate_domains_to_nodes()

    migrate_term_relationships_to_node_links()
    migrate_graph_events_to_node_events()
    
    # Step 3: Validate migration
    validate_migration()
    
    # Step 4: Drop old tables (after validation)
    drop_old_tables()
    
    # Note: Vector embeddings are automatically handled by the virtual table
    # when structure_nodes are inserted with title_embedding and definition_embedding fields populated

def migrate_layers_to_nodes():
    """Migrate layer records to structure_nodes table"""
    layers = connection.execute("SELECT * FROM layers").fetchall()
    for layer in layers:
        connection.execute("""
            INSERT INTO structure_nodes (
                id, node_type, parent_node_id, title, definition, 
                structural_predicate_id,
                title_embedding, definition_embedding,
                created_at, version, last_modified
            ) VALUES (?, 'layer', NULL, ?, ?, NULL, ?, ?, ?, ?, ?)
        """, (
            layer.id, layer.title, layer.definition,
            layer.title_embedding, layer.definition_embedding,
            layer.created_at, layer.version, layer.last_modified
        ))

def migrate_domains_to_nodes():
    """Migrate domain records to structure_nodes table"""
    domains = connection.execute("SELECT * FROM domains").fetchall()
    for domain in domains:
        connection.execute("""
            INSERT INTO structure_nodes (
                id, node_type, parent_node_id, title, definition,
                structural_predicate_id,
                title_embedding, definition_embedding,
                created_at, version, last_modified
            ) VALUES (?, 'domain', ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (
            domain.id, domain.layer_id, domain.title, domain.definition,
            domain.primary_predicate_id,
            domain.title_embedding, domain.definition_embedding,
            domain.created_at, domain.version, domain.last_modified
        ))

def migrate_terms_to_nodes():
    """Migrate term records to structure_nodes table"""
    terms = connection.execute("SELECT * FROM terms").fetchall()
    for term in terms:
        # Use parent_term_id if exists, otherwise domain_id
        parent_id = term.parent_term_id if term.parent_term_id else term.domain_id
        
        connection.execute("""
            INSERT INTO structure_nodes (
                id, node_type, parent_node_id, title, definition,
                structural_predicate_id,
                title_embedding, definition_embedding,
                created_at, version, last_modified
            ) VALUES (?, 'term', ?, ?, ?, NULL, ?, ?, ?, ?, ?)
        """, (
            term.id, parent_id, term.title, term.definition,
            term.title_embedding, term.definition_embedding,
            term.created_at, term.version, term.last_modified
        ))

def migrate_term_relationships_to_node_links():
    """Migrate term_relationships records to structure_node_links table"""
    relationships = connection.execute("SELECT * FROM term_relationships").fetchall()
    for rel in relationships:
        connection.execute("""
            INSERT INTO structure_node_links (
                id, source_node_id, target_node_id, predicate, predicate_id, created_at
            ) VALUES (?, ?, ?, ?, ?, ?)
        """, (
            rel.id, rel.source_term_id, rel.target_term_id, 
            rel.predicate, rel.predicate_id, rel.created_at
        ))

def migrate_graph_events_to_node_events():
    """Migrate graph_events records to structure_node_events table"""
    events = connection.execute("SELECT * FROM graph_events").fetchall()
    for event in events:
        # Map entity_type to node_type, with special handling for term_relationship
        if event.entity_type == 'term_relationship':
            node_type = 'structure_node_link'
        else:
            node_type = event.entity_type  # layer, domain, term
        
        connection.execute("""
            INSERT INTO structure_node_events (
                event_type, node_type, old_data, new_data, timestamp, processed
            ) VALUES (?, ?, ?, ?, ?, ?)
        """, (
            event.event_type, node_type, event.old_data, 
            event.new_data, event.timestamp, event.processed
        ))

def validate_migration():
    """Validate migration integrity"""
    # Count validation for structure_nodes
    layer_count = connection.execute("SELECT COUNT(*) FROM layers").scalar()
    domain_count = connection.execute("SELECT COUNT(*) FROM domains").scalar()
    term_count = connection.execute("SELECT COUNT(*) FROM terms").scalar()
    total_original = layer_count + domain_count + term_count
    
    node_count = connection.execute("SELECT COUNT(*) FROM structure_nodes").scalar()
    
    if node_count != total_original:
        raise MigrationError(f"Record count mismatch: {node_count} structure_nodes vs {total_original} original records")
    
    # Count validation for structure_node_links
    term_relationships_count = connection.execute("SELECT COUNT(*) FROM term_relationships").scalar()
   structure_node_links_count = connection.execute("SELECT COUNT(*) FROM structure_node_links").scalar()
    
    ifstructure_node_links_count != term_relationships_count:
        raise MigrationError(f"Link count mismatch: {node_links_count} structure_node_links vs {term_relationships_count} term_relationships")
    
    # Count validation for structure_node_events  
    graph_events_count = connection.execute("SELECT COUNT(*) FROM graph_events").scalar()
   structure_node_events_count = connection.execute("SELECT COUNT(*) FROM structure_node_events").scalar()
    
    if structure_node_events_count != graph_events_count:
        raise MigrationError(f"Event count mismatch: {node_events_count} structure_node_events vs {graph_events_count} graph_events")
    
    # Parent integrity validation
    invalid_parents = connection.execute("""
        SELECT COUNT(*) FROM structure_nodes n1
        WHERE n1.parent_node_id IS NOT NULL 
        AND NOT EXISTS (SELECT 1 FROM structure_nodes n2 WHERE n2.id = n1.parent_node_id)
    """).scalar()
    
    if invalid_parents > 0:
        raise MigrationError(f"Found {invalid_parents} structure_nodes with invalid parent references")
    
    # Validate vector embeddings virtual table is populated
    nodes_vec_count = connection.execute("SELECT COUNT(*) FROM structure_nodes_vec").scalar()
    nodes_with_embeddings_count = connection.execute("""
        SELECT COUNT(*) FROM structure_nodes 
        WHERE title_embedding IS NOT NULL OR definition_embedding IS NOT NULL
    """).scalar()
    
    if nodes_vec_count != nodes_with_embeddings_count:
        raise MigrationError(f"Vector table count mismatch: {nodes_vec_count} vs {nodes_with_embeddings_count} structure_nodes with embeddings")
```

## 3. SQLAlchemy Model Changes

### 3.1 Custom SQLAlchemy Types

**Note**: During implementation, we encountered SQLAlchemy enum conversion issues with certain versions and configurations. To resolve this, we implemented a custom SQLAlchemy type decorator that properly handles enum conversion between Python and database values.

```python
# database/enums.py - Centralized enum definitions
from enum import Enum

class NodeType(str, Enum):
    LAYER = "layer"
    DOMAIN = "domain" 
    TERM = "term"

# database/custom_types.py - Custom type decorators
from sqlalchemy.types import TypeDecorator, String
from database.enums import NodeType

class NodeTypeColumn(TypeDecorator):
    """
    Custom SQLAlchemy type for NodeType enum handling.
    
    This type decorator properly handles the conversion between
    Python NodeType enum instances and database string values,
    resolving SQLAlchemy enum conversion issues that occur with
    certain versions and configurations.
    """
    
    impl = String
    cache_ok = True
    
    def process_bind_param(self, value, dialect):
        """Convert Python enum to database string value."""
        if isinstance(value, NodeType):
            return value.value
        return value
    
    def process_result_value(self, value, dialect):
        """Convert database string value to Python enum."""
        if value is not None:
            return NodeType(value)
        return value
```

### 4.1 `NodeService` Class

```python
from enum import Enum
from typing import List, Optional, Dict, Any
from graph.graph_service import GraphService

class NodeService:
    """Centralized business logic for structure_node operations"""
    
    def __init__(self, db: Session, graph_service: GraphService):
        self.db = db
        self.graph_service = graph_service
    
    def create_node(self, node_data: Dict[str, Any]) -> StructureNode:
        """Create a new structure_node with type-specific validation"""
        node_type = NodeType(node_data['node_type'])
        
        # Validate structure_node type specific rules
        self._validate_node_creation(node_data, node_type)
        
        # Check for circular references
        if node_data.get('parent_node_id'):
            self._validate_no_circular_reference(None, node_data['parent_node_id'])
        
        # Generate embeddings
        title_embedding = generate_embedding(node_data['title'])
        definition_embedding = generate_embedding(node_data.get('definition', ''))
        
        # Create structure_node
        structure_node = StructureNode(
            node_type=node_type,
            parent_node_id=node_data.get('parent_node_id'),
            title=node_data['title'],
            definition=node_data.get('definition'),
            structural_predicate_id=node_data.get('structural_predicate_id'),
            title_embedding=title_embedding,
            definition_embedding=definition_embedding
        )
        
        self.db.add(structure_node)
        self.db.commit()
        
        # Fire event
        self._fire_node_event('create', node_type.value, None, self._node_to_dict(structure_node))
        
        return structure_node
    
    def _validate_node_creation(self, node_data: Dict[str, Any], node_type: NodeType):
        """Validate structure_node creation based on type-specific rules"""
        if node_type == NodeType.LAYER:
            self._validate_layer_creation(node_data)
        elif node_type == NodeType.DOMAIN:
            self._validate_domain_creation(node_data)
        elif node_type == NodeType.TERM:
            self._validate_term_creation(node_data)
    
    def _validate_layer_creation(self, node_data: Dict[str, Any]):
        """Validate layer creation rules"""
        # Layer titles must be unique globally
        existing = self.db.query(StructureNode).filter(
            StructureNode.node_type == NodeType.LAYER,
            StructureNode.title == node_data['title']
        ).first()
        
        if existing:
            raise ValueError("Layer title must be unique")
        
        # Layers should not have parents
        if node_data.get('parent_node_id'):
            raise ValueError("Layers cannot have parent structure_nodes")
    
    def _validate_domain_creation(self, node_data: Dict[str, Any]):
        """Validate domain creation rules"""
        parent_id = node_data.get('parent_node_id')
        if not parent_id:
            raise ValueError("Domains must have a parent layer")
        
        # Parent must be a layer
        parent = self.db.query(StructureNode).filter(StructureNode.id == parent_id).first()
        if not parent or parent.node_type != NodeType.LAYER:
            raise ValueError("Domain parent must be a layer")
        
        # Domain titles must be unique within the layer
        existing = self.db.query(StructureNode).filter(
            StructureNode.node_type == NodeType.DOMAIN,
            StructureNode.parent_node_id == parent_id,
            StructureNode.title == node_data['title']
        ).first()
        
        if existing:
            raise ValueError("Domain title must be unique within layer")
    
    def _validate_term_creation(self, node_data: Dict[str, Any]):
        """Validate term creation rules"""
        parent_id = node_data.get('parent_node_id')
        if not parent_id:
            raise ValueError("Terms must have a parent domain")
        
        # Parent must be a domain or term
        parent = self.db.query(StructureNode).filter(StructureNode.id == parent_id).first()
        if not parent or parent.node_type not in [NodeType.DOMAIN, NodeType.TERM]:
            raise ValueError("Term parent must be a domain or term")
        
        # Get the domain (either direct parent or ancestor)
        domain = self._get_domain_ancestor(parent)
        if not domain:
            raise ValueError("Term must belong to a domain")
        
        # Term titles must be unique within the domain
        existing = self._check_title_uniqueness_in_domain(domain.id, node_data['title'])
        if existing:
            raise ValueError("Term title must be unique within domain")
    
    def _validate_no_circular_reference(self, node_id: str, parent_id: str):
        """Validate no circular references using Graph Service"""
        if self.graph_service.would_create_cycle(node_id, parent_id):
            raise ValueError("Operation would create circular reference")
    
    def _get_domain_ancestor(self, structure_node: StructureNode) -> Optional[StructureNode]:
        """Get the domain ancestor of a structure_node"""
        current = structure_node
        while current:
            if current.node_type == NodeType.DOMAIN:
                return current
            current = current.parent
        return None
    
    def _check_title_uniqueness_in_domain(self, domain_id: str, title: str, exclude_id: str = None) -> bool:
        """Check if title is unique within domain using Graph Service"""
        return self.graph_service.check_title_uniqueness_in_domain(domain_id, title, exclude_id)
```

### 4.2 `NodeLinkService` Class

```python
class NodeLinkService:
    """Centralized business logic for structure_node link operations"""
    
    def __init__(self, db: Session):
        self.db = db
    
    def create_link(self, link_data: Dict[str, Any]) -> StructureNodeLink:
        """Create a new structure_node link with validation"""
        source_id = link_data['source_node_id']
        target_id = link_data['target_node_id']
        
        # Get source and target structure_nodes
        source = self.db.query(StructureNode).filter(StructureNode.id == source_id).first()
        target = self.db.query(StructureNode).filter(StructureNode.id == target_id).first()
        
        if not source or not target:
            raise ValueError("Source and target structure_nodes must exist")
        
        # Validate same structure_node type constraint
        if source.node_type != target.node_type:
            raise ValueError("StructureNodes can only be linked to structure_nodes of the same type")
        
        # Check for duplicate links
        existing = self.db.query(StructureNodeLink).filter(
            StructureNodeLink.source_node_id == source_id,
            StructureNodeLink.target_node_id == target_id,
            StructureNodeLink.predicate == link_data['predicate']
        ).first()
        
        if existing:
            raise ValueError("Link already exists")
        
        # Create link
        link = StructureNodeLink(
            source_node_id=source_id,
            target_node_id=target_id,
            predicate=link_data['predicate'],
            predicate_id=link_data.get('predicate_id')
        )
        
        self.db.add(link)
        self.db.commit()
        
        return link
```

### 5.1 New Pydantic Models

```python
from pydantic import BaseModel, Field, validator
from typing import Optional, List
from uuid import UUID
from enum import Enum

class NodeTypeEnum(str, Enum):
    LAYER = "layer"
    DOMAIN = "domain"
    TERM = "term"

class NodeBase(BaseModel):
    node_type: NodeTypeEnum
    parent_node_id: Optional[UUID] = None
    title: str = Field(..., min_length=1, max_length=255)
    definition: Optional[str] = None
    structural_predicate_id: Optional[UUID] = None

class NodeCreate(NodeBase):
    @validator('parent_node_id')
    def validate_parent_for_type(cls, v, values):
        node_type = values.get('node_type')
        if node_type == NodeTypeEnum.LAYER and v is not None:
            raise ValueError("Layers cannot have parent structure_nodes")
        if node_type in [NodeTypeEnum.DOMAIN, NodeTypeEnum.TERM] and v is None:
            raise ValueError(f"{node_type.value.title()} must have a parent structure_node")
        return v

class NodeUpdate(BaseModel):
    title: Optional[str] = Field(None, min_length=1, max_length=255)
    definition: Optional[str] = None
    parent_node_id: Optional[UUID] = None
    structural_predicate_id: Optional[UUID] = None

class NodeOut(NodeBase):
    id: UUID
    title_embedding: Optional[List[float]] = None
    definition_embedding: Optional[List[float]] = None
    created_at: str
    version: int
    last_modified: str

class NodeLinkBase(BaseModel):
    source_node_id: UUID
    target_node_id: UUID
    predicate: str
    predicate_id: Optional[UUID] = None

class NodeLinkCreate(NodeLinkBase):
    pass

class NodeLinkOut(NodeLinkBase):
    id: UUID
    created_at: str

class NodeSearchRequest(BaseModel):
    title: Optional[str] = None
    definition: Optional[str] = None
    node_type: Optional[NodeTypeEnum] = None
    parent_node_id: Optional[UUID] = None
    created_at: Optional[str] = None  # ISO8601 string
    minimum_score: Optional[float] = 0.7
    limit: Optional[int] = 10

class NodeSearchResult(NodeOut):

    distance: float

class PaginatedNodesResponse(BaseModel):
    data: List[NodeOut]
    total: int
    skip: int
    limit: int
```

### 5.2 New API Endpoints (`api/structure_nodes.py`)

```python
from fastapi import APIRouter, HTTPException, Query, Depends
from typing import List, Optional
from uuid import UUID
from sqlalchemy.orm import Session

from .models import *
from services.node_service import NodeService
from services.node_link_service import NodeLinkService

router = APIRouter()

@router.post("/", response_model=NodeOut, status_code=201)
def create_node(
    structure_node: NodeCreate, 
    db: Session = Depends(get_db),
    node_service: NodeService = Depends(get_node_service)
):
    """Create a new structure_node"""
    try:
        created_node = node_service.create_node(structure_node.dict())
        return to_node_out(created_node)
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))

@router.get("/{node_id}", response_model=NodeOut)
def get_node(
    node_id: UUID,
    db: Session = Depends(get_db)
):
    """Get a specific structure_node by ID"""
    structure_node = db.query(StructureNode).filter(StructureNode.id == str(node_id)).first()
    if not structure_node:
        raise HTTPException(status_code=404, detail="StructureNode not found")
    return to_node_out(structure_node)

@router.get("/", response_model=PaginatedNodesResponse)
def list_nodes(
    node_type: Optional[NodeTypeEnum] = None,
    parent_node_id: Optional[UUID] = None,
    skip: int = Query(0, ge=0),
    limit: int = Query(50, ge=1, le=100),
    sort_by: str = Query("title", regex="^(title|created_at)$"),
    db: Session = Depends(get_db)
):
    """List structure_nodes with filtering and pagination"""
    query = db.query(StructureNode)
    
    if node_type:
        query = query.filter(StructureNode.node_type == node_type)
    if parent_node_id:
        query = query.filter(StructureNode.parent_node_id == str(parent_node_id))
    
    total = query.count()
    
    if sort_by == "title":
        query = query.order_by(StructureNode.title)
    elif sort_by == "created_at":
        query = query.order_by(StructureNode.created_at.desc())
    
    structure_nodes = query.offset(skip).limit(limit).all()
    
    return PaginatedNodesResponse(
        data=[to_node_out(structure_node) for structure_node in structure_nodes],
        total=total,
        skip=skip,
        limit=limit
    )

@router.put("/{node_id}", response_model=NodeOut)

        return to_node_out(updated_node)
    except ValueError as e:
```

### 6.1 GraphService Modifications

```python
class GraphService:
    """Updated to work with unified structure_nodes table"""
    
    def __init__(self, db: Session):
        self.db = db
        self._build_graph()
    
    def _build_graph(self):
        """Build graph structure from structure_nodes table"""
        self.structure_nodes = {}
        self.edges = {}
        
        # Load all structure_nodes
        structure_nodes = self.db.query(StructureNode).all()
        for structure_node in structure_nodes:
            self.structure_nodes[structure_node.id] = {
                'id': structure_node.id,
                'type': structure_node.node_type.value,
                'title': structure_node.title,
                'parent_id': structure_node.parent_node_id,
                'children': []
            }
        
        # Build parent-child relationships
        for node_id, node_data in self.structure_nodes.items():
            if node_data['parent_id']:
                parent = self.structure_nodes.get(node_data['parent_id'])
                if parent:
                    parent['children'].append(node_id)
        
        # Load structure_node links
        links = self.db.query(StructureNodeLink).all()
        for link in links:
            source_id = link.source_node_id
            if source_id not in self.edges:
                self.edges[source_id] = []
            self.edges[source_id].append({
                'target_id': link.target_node_id,
                'predicate': link.predicate
            })
    
    def would_create_cycle(self, node_id: str, new_parent_id: str) -> bool:
        """Check if setting new_parent_id as parent of node_id would create a cycle"""
        if not node_id or not new_parent_id:
            return False
        
        # Walk up the ancestor chain from new_parent_id
        current = new_parent_id
        visited = set()
        
        while current and current not in visited:
            if current == node_id:
                return True
            visited.add(current)
            node_data = self.structure_nodes.get(current)
            current = node_data['parent_id'] if node_data else None
        
        return False
    
    def check_title_uniqueness_in_domain(self, domain_id: str, title: str, exclude_id: str = None) -> bool:
        """Check if title is unique within domain using graph traversal"""
        # Get all structure_nodes in the domain subtree
        domain_nodes = self._get_domain_subtree(domain_id)
        
        for node_id in domain_nodes:
            node_data = self.structure_nodes.get(node_id)
            if (node_data and 
                node_data['title'] == title and 
                node_id != exclude_id and
                node_data['type'] == 'term'):
                return True
        
        return False
    
    def _get_domain_subtree(self, domain_id: str) -> List[str]:
        """Get all structure_nodes in a domain's subtree"""
        result = []
        stack = [domain_id]
        
        while stack:
            current_id = stack.pop()
            result.append(current_id)
            
            node_data = self.structure_nodes.get(current_id)
            if node_data:
                stack.extend(node_data['children'])
        
        return result
```

```python
class NodeEventHandler:
    """Handle structure_node events and maintain backward compatibility"""
    
    def __init__(self, db: Session):
        self.db = db
    
    def fire_node_event(self, event_type: str, node_type: str, old_data: dict, new_data: dict):
        """Fire unified structure_node event and legacy events for backward compatibility"""
        # Create unified structure_node event
        event = NodeEvent(
            event_type=event_type,
            node_type=node_type,
            old_data=old_data,
            new_data=new_data
        )
        self.db.add(event)
        
        # Fire legacy events for backward compatibility
        if node_type in ['layer', 'domain', 'term']:
            legacy_event = GraphEvent(
                event_type=event_type,
                entity_type=node_type,
                old_data=old_data,
                new_data=new_data
            )
            self.db.add(legacy_event)
        
        self.db.commit()
```
