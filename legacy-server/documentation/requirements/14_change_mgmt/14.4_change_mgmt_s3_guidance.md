# S3 for user verification

Utilize an S3 based user verification process to avoid needing an email service:

```python
class S3OnlyVerification:
    """Verification without email - users share verification codes out-of-band"""
    
    def initiate_registration(self, username: str) -> str:
        """Step 1: User picks username, gets pending ID"""
        pending_id = hashlib.sha256(username.encode()).hexdigest()[:12]
        
        # Write pending registration
        self.s3.put_object(
            Bucket=self.bucket,
            Key=f"users/pending/{pending_id}.json",
            Body=json.dumps({
                "username": username,
                "pending_id": pending_id,
                "created_at": datetime.utcnow().isoformat(),
                "verified_by": []  # Needs N colleagues to verify
            })
        )
        
        return pending_id
    
    def verify_colleague(self, pending_id: str, verifier_id: str):
        """Step 2: Existing users verify new user"""
        # Fetch pending user
        obj = self.s3.get_object(
            Bucket=self.bucket,
            Key=f"users/pending/{pending_id}.json"
        )
        pending_user = json.loads(obj['Body'].read())
        
        # Add verification
        pending_user['verified_by'].append({
            "verifier_id": verifier_id,
            "verified_at": datetime.utcnow().isoformat()
        })
        
        # If enough verifications, activate
        if len(pending_user['verified_by']) >= 2:  # Require 2 colleagues
            self.activate_user(pending_user)
        else:
            # Update pending
            self.s3.put_object(
                Bucket=self.bucket,
                Key=f"users/pending/{pending_id}.json",
                Body=json.dumps(pending_user)
            )
    
    def activate_user(self, user_data):
        """Move from pending to active"""
        # Create active user
        self.s3.put_object(
            Bucket=self.bucket,
            Key=f"users/active/{user_data['username']}.json",
            Body=json.dumps({
                **user_data,
                "status": "active",
                "activated_at": datetime.utcnow().isoformat()
            })
        )
        
        # Delete pending
        self.s3.delete_object(
            Bucket=self.bucket,
            Key=f"users/pending/{user_data['pending_id']}.json"
        )
```

# Avoid vendor lock-in

Ensure that all S3 integration in compatible with non-amazon S3 compatible services.

Excellent point about vendor lock-in! The S3 API has become a de facto standard, and there are many S3-compatible alternatives. Let me break down your options:

## S3-Compatible Storage Options

### 1. **MinIO** (Self-Hosted)
The most popular open-source S3-compatible storage:

```python
# Works identically with MinIO
import duckdb
import boto3

# MinIO connection
s3_client = boto3.client(
    's3',
    endpoint_url='https://minio.yourserver.com:9000',
    aws_access_key_id='minioadmin',
    aws_secret_access_key='minioadmin'
)

# DuckDB with MinIO
conn = duckdb.connect()
conn.execute("""
    SET s3_endpoint='minio.yourserver.com:9000';
    SET s3_access_key_id='minioadmin';
    SET s3_secret_access_key='minioadmin';
    SET s3_use_ssl=true;
    SET s3_url_style='path';  -- Important for MinIO
""")

# Now works exactly like S3
conn.execute("SELECT * FROM read_parquet('s3://bucket/file.parquet')")
```

## Vendor-Agnostic Implementation

Here's how to make your code work with ANY S3-compatible storage:

```python
import os
from dataclasses import dataclass
from typing import Optional
import boto3
import duckdb

@dataclass
class S3Config:
    """S3-compatible storage configuration"""
    endpoint_url: Optional[str] = None  # None = AWS S3
    access_key_id: str = ""
    secret_access_key: str = ""
    bucket_name: str = ""
    region: str = "us-east-1"
    use_ssl: bool = True
    url_style: str = "path"  # 'path' or 'virtual-host'
    
    @classmethod
    def from_env(cls):
        """Load from environment variables"""
        return cls(
            endpoint_url=os.getenv('S3_ENDPOINT_URL'),
            access_key_id=os.getenv('S3_ACCESS_KEY_ID', ''),
            secret_access_key=os.getenv('S3_SECRET_ACCESS_KEY', ''),
            bucket_name=os.getenv('S3_BUCKET_NAME', ''),
            region=os.getenv('S3_REGION', 'us-east-1'),
            use_ssl=os.getenv('S3_USE_SSL', 'true').lower() == 'true',
            url_style=os.getenv('S3_URL_STYLE', 'path')
        )

class VendorAgnosticStorage:
    def __init__(self, config: S3Config):
        self.config = config
        
        # boto3 client works with all S3-compatible services
        self.s3_client = boto3.client(
            's3',
            endpoint_url=config.endpoint_url,
            aws_access_key_id=config.access_key_id,
            aws_secret_access_key=config.secret_access_key,
            region_name=config.region,
            use_ssl=config.use_ssl
        )
        
    def setup_duckdb(self, conn: duckdb.DuckDBPyConnection):
        """Configure DuckDB for S3-compatible storage"""
        conn.execute("INSTALL httpfs; LOAD httpfs;")
        
        if self.config.endpoint_url:
            # Parse endpoint for DuckDB
            endpoint = self.config.endpoint_url.replace('https://', '').replace('http://', '')
            conn.execute(f"SET s3_endpoint='{endpoint}';")
            
        conn.execute(f"SET s3_access_key_id='{self.config.access_key_id}';")
        conn.execute(f"SET s3_secret_access_key='{self.config.secret_access_key}';")
        conn.execute(f"SET s3_region='{self.config.region}';")
        conn.execute(f"SET s3_use_ssl={str(self.config.use_ssl).lower()};")
        conn.execute(f"SET s3_url_style='{self.config.url_style}';")
        
    def write_changes(self, changes: list, timestamp: str):
        """Write changes to S3-compatible storage"""
        key = f"changes/{timestamp}.json"
        self.s3_client.put_object(
            Bucket=self.config.bucket_name,
            Key=key,
            Body=json.dumps(changes)
        )
        
    def read_with_duckdb(self, conn: duckdb.DuckDBPyConnection):
        """Read directly from S3-compatible storage"""
        return conn.execute(f"""
            SELECT * FROM read_parquet('s3://{self.config.bucket_name}/changes/*.parquet')
        """).fetchall()
```

## Identity Registration with S3-Compatible Storage

The identity system works identically across all providers:

```python
class S3CompatibleIdentity:
    def __init__(self, storage: VendorAgnosticStorage):
        self.storage = storage
        
    def register_user(self, email: str, name: str) -> str:
        """Works with any S3-compatible storage"""
        user_id = hashlib.sha256(email.encode()).hexdigest()[:16]
        token = secrets.token_urlsafe(32)
        
        registration = {
            "user_id": user_id,
            "email": email,
            "name": name,
            "token": token,
            "status": "pending",
            "created_at": datetime.utcnow().isoformat()
        }
        
        # This works with ANY S3-compatible storage
        self.storage.s3_client.put_object(
            Bucket=self.storage.config.bucket_name,
            Key=f"users/pending/{user_id}.json",
            Body=json.dumps(registration)
        )
        
        return token
```

## Recommendation

For your use case, I'd suggest:

1. **Development**: MinIO locally via Docker
2. **Production** (pick one):
   - **Cloudflare R2**: Best for high-traffic (no egress fees)
   - **Backblaze B2**: Best for large storage (cheapest storage)
   - **MinIO on VPS**: Best for control (one-time cost)

All these options work identically with your code - just change the endpoint URL and credentials. This gives you complete vendor independence while using the same S3 API everywhere.

The beauty is that DuckDB, boto3, and most tools already support S3-compatible APIs, so you're using a true industry standard, not just AWS's implementation.
