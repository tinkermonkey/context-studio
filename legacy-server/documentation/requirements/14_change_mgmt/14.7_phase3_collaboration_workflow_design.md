# Phase 3: Collaboration Workflow Design Document

## 1. Overview

### 1.1 Purpose
Design and implement the collaborative change management workflow that enables teams to propose, review, vote on, and merge changes in a distributed Git-like system using S3 as the shared storage layer.

### 1.2 Scope
This phase builds upon Phase 1 and 2 by implementing:
- Changeset creation and storage in S3 as Parquet files
- Proposal system with metadata and voting stored in S3
- Voting mechanism via S3 vote files with consensus tracking
- Auto-merge approved changes with CRDT conflict resolution
- Lightweight email-based identity verification system

### 1.3 Success Criteria
- Users can bundle changes into named changesets for collaboration
- Teams can create and review proposals with full change visibility
- Voting system enables democratic decision-making with configurable thresholds
- Approved changes merge automatically with conflict resolution
- Identity system enables secure collaboration without complex infrastructure

#### 2.1.2 Changeset Manager
```python
class ChangesetManager:
    """Manages changeset creation, modification, and lifecycle."""
    
    def __init__(self, sqlite_conn, s3_sync_manager, working_tree_manager):
        self.sqlite_conn = sqlite_conn
        self.s3_sync = s3_sync_manager
        self.working_tree = working_tree_manager
    
    def create_changeset(self, title: str, description: str, author_id: str,
                        include_staged: bool = True) -> Changeset:
        """Create a new changeset from staged or all working changes."""
        
        changeset_id = str(uuid.uuid4())
        
        # Get changes to include
        if include_staged:
            changes = self.working_tree.get_staged_changes()
        else:
            changes = self.working_tree.get_working_changes()
        
        if not changes:
            raise ValueError("No changes available to create changeset")
        
        # Create changeset record
        changeset = Changeset(
            id=changeset_id,
            title=title,
            description=description,
            state=ChangesetState.DRAFT,
            branch_name=None,
            parent_changeset_id=None,
            author_id=author_id,
            created_at=datetime.now(),
            metadata={"changes_count": len(changes)}
        )
        
        # Store changeset locally
        self._store_changeset_locally(changeset, changes)
        
        # Update version records with changeset_id
        self._associate_versions_with_changeset(changeset_id, changes)
        
        return changeset
    
    def get_changeset(self, changeset_id: str) -> Optional[Changeset]:
        """Retrieve changeset by ID."""
        
        query = """
        SELECT * FROM changesets WHERE id = ?
        """
        
        result = self.sqlite_conn.execute(query, (changeset_id,)).fetchone()
        if not result:
            return None
            
        return self._row_to_changeset(result)
    
    def list_changesets(self, author_id: Optional[str] = None, 
                       state: Optional[ChangesetState] = None) -> List[Changeset]:
        """List changesets with optional filtering."""
        
        query = "SELECT * FROM changesets WHERE 1=1"
        params = []
        
        if author_id:
            query += " AND author_id = ?"
            params.append(author_id)
            
        if state:
            query += " AND state = ?"
            params.append(state.value)
            
        query += " ORDER BY created_at DESC"
        
        results = self.sqlite_conn.execute(query, params).fetchall()
        return [self._row_to_changeset(row) for row in results]
    
    def update_changeset(self, changeset_id: str, title: Optional[str] = None,
                        description: Optional[str] = None) -> bool:
        """Update changeset metadata."""
        
        updates = []
        params = []
        
        if title:
            updates.append("title = ?")
            params.append(title)
            
        if description:
            updates.append("description = ?")
            params.append(description)
            
        if not updates:
            return False
            
        params.append(changeset_id)
        query = f"UPDATE changesets SET {', '.join(updates)} WHERE id = ?"
        
        self.sqlite_conn.execute(query, params)
        self.sqlite_conn.commit()
        return True
    
    def _store_changeset_locally(self, changeset: Changeset, changes: List):
        """Store changeset in local SQLite database."""
        
        query = """
        INSERT INTO changesets (
            id, title, description, state, branch_name, parent_changeset_id,
            author_id, created_at, merged_at, metadata
        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """
        
        params = (
            changeset.id, changeset.title, changeset.description,
            changeset.state.value, changeset.branch_name, changeset.parent_changeset_id,
            changeset.author_id, changeset.created_at.isoformat(),
            changeset.merged_at.isoformat() if changeset.merged_at else None,
            json.dumps(changeset.metadata) if changeset.metadata else None
        )
        
        self.sqlite_conn.execute(query, params)
        self.sqlite_conn.commit()
```

```python
class ProposalManager:
    """Manages proposal creation, voting, and lifecycle."""
    
    def __init__(self, sqlite_conn, s3_sync_manager, changeset_manager):
        self.sqlite_conn = sqlite_conn
        self.s3_sync = s3_sync_manager
        self.changeset_manager = changeset_manager
        
    def create_proposal(self, changeset_id: str, title: str, description: str,
                       created_by: str, required_approvals: int = 1) -> Proposal:
        """Create a new proposal for changeset review."""
        
        # Validate changeset exists and is in appropriate state
        changeset = self.changeset_manager.get_changeset(changeset_id)
        if not changeset:
            raise ValueError(f"Changeset {changeset_id} not found")
            
        if changeset.state not in [ChangesetState.DRAFT, ChangesetState.PROPOSED]:
            raise ValueError(f"Cannot create proposal for changeset in {changeset.state} state")
        
        proposal_id = str(uuid.uuid4())
        
        proposal = Proposal(
            id=proposal_id,
            changeset_id=changeset_id,
            title=title,
            description=description,
            status=ProposalStatus.OPEN,
            required_approvals=required_approvals,
            created_by=created_by,
            created_at=datetime.now()
        )
        
        # Store proposal locally
        self._store_proposal_locally(proposal)
        
        # Update changeset state
        self._update_changeset_state(changeset_id, ChangesetState.PROPOSED)
        
        # Push proposal to S3
        self._push_proposal_to_s3(proposal)
        
        return proposal
    
    def vote_on_proposal(self, proposal_id: str, user_id: str, vote: str,
                        comment: Optional[str] = None) -> ProposalVote:
        """Cast vote on a proposal."""
        
        if vote not in ['approve', 'reject', 'abstain']:
            raise ValueError("Vote must be 'approve', 'reject', or 'abstain'")
        
        # Check if proposal exists and is open
        proposal = self.get_proposal(proposal_id)
        if not proposal:
            raise ValueError(f"Proposal {proposal_id} not found")
            
        if proposal.status != ProposalStatus.OPEN:
            raise ValueError(f"Cannot vote on proposal with status {proposal.status}")
        
        # Create vote record
        proposal_vote = ProposalVote(
            proposal_id=proposal_id,
            user_id=user_id,
            vote=vote,
            comment=comment,
            voted_at=datetime.now()
        )
        
        # Store vote locally
        self._store_vote_locally(proposal_vote)
        
        # Push vote to S3
        self._push_vote_to_s3(proposal_vote)
        
        # Check if proposal should be automatically approved/rejected
        self._evaluate_proposal_status(proposal_id)
        
        return proposal_vote
    
    def get_proposal(self, proposal_id: str) -> Optional[Proposal]:
        """Retrieve proposal by ID."""
        
        query = "SELECT * FROM proposals WHERE id = ?"
        result = self.sqlite_conn.execute(query, (proposal_id,)).fetchone()
        
        if not result:
            return None
            
        return self._row_to_proposal(result)
    
    def list_proposals(self, status: Optional[ProposalStatus] = None,
                      created_by: Optional[str] = None) -> List[Proposal]:
        """List proposals with optional filtering."""
        
        query = "SELECT * FROM proposals WHERE 1=1"
        params = []
        
        if status:
            query += " AND status = ?"
            params.append(status.value)
            
        if created_by:
            query += " AND created_by = ?"
            params.append(created_by)
            
        query += " ORDER BY created_at DESC"
        
        results = self.sqlite_conn.execute(query, params).fetchall()
        return [self._row_to_proposal(row) for row in results]
    
    def get_proposal_votes(self, proposal_id: str) -> List[ProposalVote]:
        """Get all votes for a proposal."""
        
        query = """
        SELECT * FROM proposal_votes 
        WHERE proposal_id = ? 
        ORDER BY voted_at
        """
        
        results = self.sqlite_conn.execute(query, (proposal_id,)).fetchall()
        return [self._row_to_vote(row) for row in results]
    
    def _evaluate_proposal_status(self, proposal_id: str):
        """Evaluate whether proposal should change status based on votes."""
        
        proposal = self.get_proposal(proposal_id)
        votes = self.get_proposal_votes(proposal_id)
        
        approve_votes = len([v for v in votes if v.vote == 'approve'])
        reject_votes = len([v for v in votes if v.vote == 'reject'])
        
        # Auto-approve if threshold met
        if approve_votes >= proposal.required_approvals:
            self._update_proposal_status(proposal_id, ProposalStatus.APPROVED)
            # Trigger auto-merge if configured
            self._auto_merge_if_enabled(proposal_id)
            
        # Auto-reject if more rejections than possible approvals
        # (This is a simple policy, could be made configurable)
        elif reject_votes > approve_votes and reject_votes >= proposal.required_approvals:
            self._update_proposal_status(proposal_id, ProposalStatus.REJECTED)
    
    def _push_proposal_to_s3(self, proposal: Proposal):
        """Push proposal metadata to S3."""
        
        proposal_data = {
            "id": proposal.id,
            "changeset_id": proposal.changeset_id,
            "title": proposal.title,
            "description": proposal.description,
            "status": proposal.status.value,
            "required_approvals": proposal.required_approvals,
            "created_by": proposal.created_by,
            "created_at": proposal.created_at.isoformat(),
            "closed_at": proposal.closed_at.isoformat() if proposal.closed_at else None
        }
        
        s3_path = f"s3://{self.s3_sync.s3_config['bucket']}/proposals/{proposal.id}/metadata.parquet"
        
        df = pd.DataFrame([proposal_data])
        self.s3_sync.parquet_writer.write_metadata(proposal_data, s3_path)
    
    def _push_vote_to_s3(self, vote: ProposalVote):
        """Push vote to S3."""
        
        vote_data = {
            "proposal_id": vote.proposal_id,
            "user_id": vote.user_id,
            "vote": vote.vote,
            "comment": vote.comment,
            "voted_at": vote.voted_at.isoformat()
        }
        
        s3_path = f"s3://{self.s3_sync.s3_config['bucket']}/proposals/{vote.proposal_id}/votes/vote_{vote.user_id}.parquet"
        
        self.s3_sync.parquet_writer.write_metadata(vote_data, s3_path)
```

### 2.3 CRDT-Based Auto-Merge System

#### 2.3.1 CRDT Merge Engine
```python
class CRDTMergeEngine:
    """Handles conflict-free replicated data type merging for approved changes."""
    
    def __init__(self, version_manager, entity_manager):
        self.version_manager = version_manager
        self.entity_manager = entity_manager
    
    def merge_changeset(self, changeset_id: str, merge_author: str) -> Dict[str, Any]:
        """Merge approved changeset using CRDT semantics."""
        
        changeset = self.changeset_manager.get_changeset(changeset_id)
        if not changeset:
            raise ValueError(f"Changeset {changeset_id} not found")
            
        # Get all versions in changeset
        changeset_versions = self._get_changeset_versions(changeset_id)
        
        # Group by entity for merge processing
        entities_to_merge = {}
        for version in changeset_versions:
            key = (version.entity_type, version.entity_id)
            if key not in entities_to_merge:
                entities_to_merge[key] = []
            entities_to_merge[key].append(version)
        
        merge_results = []
        conflicts = []
        
        # Process each entity
        for (entity_type, entity_id), versions in entities_to_merge.items():
            
            try:
                # Get current canonical version
                canonical = self.version_manager.get_canonical_version(entity_type, entity_id)
                
                # Apply CRDT merge for each version
                merged_content = self._apply_crdt_merge(canonical, versions)
                
                # Create merged version
                merged_version = self.version_manager.create_version(
                    entity_type=entity_type,
                    entity_id=entity_id,
                    content=merged_content,
                    author_id=merge_author,
                    state=ChangeState.MERGED
                )
                
                # Update canonical reference
                self._update_canonical_version(entity_type, entity_id, merged_version.id)
                
                merge_results.append({
                    "entity_type": entity_type,
                    "entity_id": entity_id,
                    "merged_version_id": merged_version.id,
                    "status": "success"
                })
                
            except CRDTConflictError as e:
                conflicts.append({
                    "entity_type": entity_type,
                    "entity_id": entity_id,
                    "error": str(e),
                    "requires_manual_resolution": True
                })
        
        # Update changeset state
        if not conflicts:
            self._update_changeset_state(changeset_id, ChangesetState.MERGED)
        
        return {
            "changeset_id": changeset_id,
            "merged_entities": len(merge_results),
            "conflicts": len(conflicts),
            "results": merge_results,
            "conflict_details": conflicts
        }
    
    def _apply_crdt_merge(self, canonical_version: Optional[EntityVersion], 
                         changeset_versions: List[EntityVersion]) -> Dict[str, Any]:
        """Apply CRDT merge logic to combine versions."""
        
        if not canonical_version:
            # New entity, use latest version
            return changeset_versions[-1].content
        
        base_content = canonical_version.content
        
        # Apply each changeset version in order
        merged_content = base_content.copy()
        
        for version in changeset_versions:
            merged_content = self._merge_content_crdt(merged_content, version.content)
        
        return merged_content
    
    def _merge_content_crdt(self, base: Dict[str, Any], changes: Dict[str, Any]) -> Dict[str, Any]:
        """Merge two content objects using CRDT semantics."""
        
        # This is a simplified CRDT implementation
        # In practice, you'd want more sophisticated conflict resolution
        
        result = base.copy()
        
        for key, value in changes.items():
            if key not in result:
                # New field, add it
                result[key] = value
            elif isinstance(value, dict) and isinstance(result[key], dict):
                # Nested object, recursively merge
                result[key] = self._merge_content_crdt(result[key], value)
            elif isinstance(value, list) and isinstance(result[key], list):
                # List merge - union with deduplication
                combined = result[key] + [item for item in value if item not in result[key]]
                result[key] = combined
            else:
                # Scalar conflict - use last-writer-wins (could be enhanced)
                result[key] = value
        
        return result
```

### 2.4 Identity and Trust System

#### 2.4.1 Lightweight Identity Manager
```python
@dataclass
class UserIdentity:
    user_id: str
    email: str
    display_name: str
    public_key: Optional[str]
    verified: bool
    trust_level: int  # 0=unverified, 1=email verified, 2=team verified
    created_at: datetime
    verified_at: Optional[datetime] = None

class IdentityManager:
    """Manages user identity and trust verification."""
    
    def __init__(self, sqlite_conn, s3_sync_manager):
        self.sqlite_conn = sqlite_conn
        self.s3_sync = s3_sync_manager
    
    def register_user(self, email: str, display_name: str) -> UserIdentity:
        """Register new user with email-based identity."""
        
        user_id = self._generate_user_id(email)
        
        # Check if user already exists
        existing = self.get_user(user_id)
        if existing:
            return existing
        
        # Create new user identity
        identity = UserIdentity(
            user_id=user_id,
            email=email,
            display_name=display_name,
            public_key=None,
            verified=False,
            trust_level=0,
            created_at=datetime.now()
        )
        
        # Store locally
        self._store_identity_locally(identity)
        
        # Send verification code
        verification_code = self._generate_verification_code()
        self._store_verification_code(user_id, verification_code)
        
        # In a real system, you'd send an email here
        # For this implementation, we'll store the code for retrieval
        
        return identity
    
    def verify_email(self, user_id: str, verification_code: str) -> bool:
        """Verify user email with verification code."""
        
        stored_code = self._get_verification_code(user_id)
        if not stored_code or stored_code != verification_code:
            return False
        
        # Update user verification status
        query = """
        UPDATE user_identities 
        SET verified = 1, trust_level = 1, verified_at = ?
        WHERE user_id = ?
        """
        
        self.sqlite_conn.execute(query, (datetime.now().isoformat(), user_id))
        self.sqlite_conn.commit()
        
        # Clean up verification code
        self._delete_verification_code(user_id)
        
        return True
    
    def trust_user(self, trustee_user_id: str, trusted_user_id: str) -> bool:
        """Allow one verified user to vouch for another (peer verification)."""
        
        trustee = self.get_user(trustee_user_id)
        if not trustee or trustee.trust_level < 1:
            return False  # Only verified users can vouch for others
        
        # Record trust relationship
        query = """
        INSERT OR REPLACE INTO user_trust_relationships 
        (trustee_user_id, trusted_user_id, created_at)
        VALUES (?, ?, ?)
        """
        
        self.sqlite_conn.execute(query, (
            trustee_user_id, trusted_user_id, datetime.now().isoformat()
        ))
        
        # Check if user now has enough trust to be team-verified
        trust_count = self._count_trust_relationships(trusted_user_id)
        
        if trust_count >= 2:  # Require 2 team members to vouch
            query = """
            UPDATE user_identities 
            SET trust_level = 2 
            WHERE user_id = ?
            """
            self.sqlite_conn.execute(query, (trusted_user_id,))
        
        self.sqlite_conn.commit()
        return True
    
    def get_user(self, user_id: str) -> Optional[UserIdentity]:
        """Get user identity by ID."""
        
        query = "SELECT * FROM user_identities WHERE user_id = ?"
        result = self.sqlite_conn.execute(query, (user_id,)).fetchone()
        
        if not result:
            return None
            
        return self._row_to_identity(result)
    
    def _generate_user_id(self, email: str) -> str:
        """Generate deterministic user ID from email."""
        import hashlib
        return hashlib.sha256(email.lower().encode()).hexdigest()[:16]
    
    def _generate_verification_code(self) -> str:
        """Generate 6-digit verification code."""
        import random
        return f"{random.randint(100000, 999999)}"
```
