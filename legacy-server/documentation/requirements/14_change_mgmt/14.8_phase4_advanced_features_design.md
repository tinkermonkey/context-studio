# Phase 4: Advanced Features Design Document

## 1. Overview

### 1.1 Purpose
Design and implement advanced collaborative features that provide sophisticated change management capabilities including branch management, intelligent conflict resolution, analytical views, and incremental synchronization for enterprise-scale deployments.

### 1.2 Scope
This phase builds upon Phases 1-3 by implementing:
- Branch management in S3 structure with isolated development workflows
- Advanced conflict resolution with CRDT semantics and manual resolution UI
- DuckDB analytical views for comprehensive change history analysis
- Incremental sync with partitioned queries for efficient large-scale operations

### 1.3 Success Criteria
- Teams can create and manage multiple development branches with isolation
- Complex merge conflicts are detected and resolved automatically or through guided UI
- Analytical queries provide insights into team collaboration patterns and change metrics
- Large datasets sync efficiently using incremental and partitioned approaches
- Enterprise-scale teams (50+ users, 10K+ entities) can collaborate effectively

from typing import List, Optional, Dict, Any, Set
from datetime import datetime

class BranchType(Enum):

class Branch:

```python
class BranchManager:
    """Manages branch creation, switching, and merging operations."""
    
    def __init__(self, sqlite_conn, s3_sync_manager, changeset_manager):
        self.sqlite_conn = sqlite_conn
        self.s3_sync = s3_sync_manager
        self.changeset_manager = changeset_manager
        
    def create_branch(self, name: str, branch_type: BranchType, 
                     base_branch_id: Optional[str], created_by: str,
                     description: Optional[str] = None) -> Branch:
        """Create a new development branch."""
        
        branch_id = str(uuid.uuid4())
        
        # Get base branch head changeset
        if base_branch_id:
            base_branch = self.get_branch(base_branch_id)
            if not base_branch:
                raise ValueError(f"Base branch {base_branch_id} not found")
            head_changeset_id = base_branch.head_changeset_id
        else:
            # Creating main branch or orphaned branch
            head_changeset_id = None
        
        branch = Branch(
            id=branch_id,
            name=name,
            branch_type=branch_type,
            base_branch_id=base_branch_id,
            head_changeset_id=head_changeset_id,
            created_by=created_by,
            created_at=datetime.now(),
            description=description
        )
        
        # Store branch locally
        self._store_branch_locally(branch)
        
        # Create S3 branch structure
        self._create_s3_branch_structure(branch)
        
        return branch
    
    def switch_branch(self, branch_id: str, user_id: str) -> Dict[str, Any]:
        """Switch working tree to different branch."""
        
        branch = self.get_branch(branch_id)
        if not branch:
            raise ValueError(f"Branch {branch_id} not found")
        
        # Check for uncommitted changes
        working_changes = self.working_tree.get_working_changes()
        if working_changes:
            return {
                "status": "error",
                "message": "Cannot switch branches with uncommitted changes",
                "uncommitted_changes": len(working_changes)
            }
        
        # Update working tree to branch head
        if branch.head_changeset_id:
            self._update_working_tree_to_changeset(branch.head_changeset_id)
        
        # Update current branch reference
        self._set_current_branch(user_id, branch_id)
        
        return {
            "status": "success",
            "message": f"Switched to branch '{branch.name}'",
            "branch": branch
        }
    
    def create_merge_request(self, source_branch_id: str, target_branch_id: str,
                           title: str, description: str, created_by: str) -> BranchMergeRequest:
        """Create merge request between branches."""
        
        source_branch = self.get_branch(source_branch_id)
        target_branch = self.get_branch(target_branch_id)
        
        if not source_branch or not target_branch:
            raise ValueError("Source or target branch not found")
        
        # Analyze conflicts
        conflicts = self._analyze_branch_conflicts(source_branch, target_branch)
        
        merge_request = BranchMergeRequest(
            id=str(uuid.uuid4()),
            source_branch_id=source_branch_id,
            target_branch_id=target_branch_id,
            title=title,
            description=description,
            created_by=created_by,
            status='open',
            conflicts=conflicts,
            created_at=datetime.now()
        )
        
        # Store merge request
        self._store_merge_request_locally(merge_request)
        
        # Push to S3 for collaboration
        self._push_merge_request_to_s3(merge_request)
        
        return merge_request
    
    def merge_branches(self, merge_request_id: str, merge_strategy: str = 'auto') -> Dict[str, Any]:
        """Merge source branch into target branch."""
        
        merge_request = self.get_merge_request(merge_request_id)
        if not merge_request:
            raise ValueError(f"Merge request {merge_request_id} not found")
        
        if merge_request.status != 'approved':
            raise ValueError(f"Merge request must be approved before merging")
        
        # Get branch changesets to merge
        source_changesets = self._get_branch_changesets(merge_request.source_branch_id)
        
        # Apply merge strategy
        if merge_strategy == 'auto' and not merge_request.conflicts:
            return self._auto_merge_branches(merge_request, source_changesets)
        elif merge_strategy == 'manual':
            return self._manual_merge_branches(merge_request, source_changesets)
        else:
            return {
                "status": "error",
                "message": "Conflicts detected, manual merge required",
                "conflicts": merge_request.conflicts
            }
    
    def _analyze_branch_conflicts(self, source_branch: Branch, target_branch: Branch) -> List[Dict[str, Any]]:
        """Analyze potential conflicts between branches."""
        
        # Get changesets unique to each branch
        source_changesets = self._get_branch_changesets(source_branch.id)
        target_changesets = self._get_branch_changesets(target_branch.id)
        
        conflicts = []
        
        # Find entities modified in both branches
        source_entities = set()
        target_entities = set()
        
        for changeset in source_changesets:
            entities = self._get_changeset_entities(changeset.id)
            source_entities.update(entities)
            
        for changeset in target_changesets:
            entities = self._get_changeset_entities(changeset.id)
            target_entities.update(entities)
        
        # Conflicting entities
        conflicting_entities = source_entities.intersection(target_entities)
        
        for entity_key in conflicting_entities:
            entity_type, entity_id = entity_key
            
            conflicts.append({
                "entity_type": entity_type,
                "entity_id": entity_id,
                "conflict_type": "concurrent_modification",
                "source_versions": self._get_entity_versions_in_branch(source_branch.id, entity_type, entity_id),
                "target_versions": self._get_entity_versions_in_branch(target_branch.id, entity_type, entity_id)
            })
        
        return conflicts
```

### 2.2 Advanced Conflict Resolution

#### 2.2.1 Intelligent Conflict Detector
```python
class ConflictType(Enum):
    CONCURRENT_MODIFICATION = "concurrent_modification"
    STRUCTURAL_CONFLICT = "structural_conflict"
    DEPENDENCY_CONFLICT = "dependency_conflict"
    SEMANTIC_CONFLICT = "semantic_conflict"

@dataclass
class ConflictDescriptor:
    conflict_id: str
    conflict_type: ConflictType
    entity_type: str
    entity_id: str
    local_version: EntityVersion
    remote_version: EntityVersion
    conflict_details: Dict[str, Any]
    resolution_suggestions: List[Dict[str, Any]]
    severity: str  # 'low', 'medium', 'high'

class IntelligentConflictDetector:
    """Advanced conflict detection with semantic analysis."""
    
    def __init__(self, version_manager, nlp_analyzer=None):
        self.version_manager = version_manager
        self.nlp_analyzer = nlp_analyzer  # Optional NLP for semantic analysis
    
    def detect_conflicts(self, local_versions: List[EntityVersion], 
                        remote_versions: List[EntityVersion]) -> List[ConflictDescriptor]:
        """Detect and categorize conflicts between version sets."""
        
        conflicts = []
        
        # Group versions by entity
        local_by_entity = self._group_versions_by_entity(local_versions)
        remote_by_entity = self._group_versions_by_entity(remote_versions)
        
        # Find entities with conflicting versions
        conflicting_entities = set(local_by_entity.keys()) & set(remote_by_entity.keys())
        
        for entity_key in conflicting_entities:
            local_version = local_by_entity[entity_key][-1]  # Latest local version
            remote_version = remote_by_entity[entity_key][-1]  # Latest remote version
            
            conflict_descriptor = self._analyze_version_conflict(local_version, remote_version)
            if conflict_descriptor:
                conflicts.append(conflict_descriptor)
        
        return conflicts
    
    def _analyze_version_conflict(self, local_version: EntityVersion, 
                                 remote_version: EntityVersion) -> Optional[ConflictDescriptor]:
        """Analyze specific version conflict and generate resolution suggestions."""
        
        # Compare content structures
        structural_conflicts = self._detect_structural_conflicts(local_version.content, remote_version.content)
        
        # Check for dependency conflicts
        dependency_conflicts = self._detect_dependency_conflicts(local_version, remote_version)
        
        # Semantic analysis if NLP available
        semantic_conflicts = []
        if self.nlp_analyzer:
            semantic_conflicts = self._detect_semantic_conflicts(local_version, remote_version)
        
        # Determine primary conflict type
        if semantic_conflicts:
            conflict_type = ConflictType.SEMANTIC_CONFLICT
            severity = "high"
        elif dependency_conflicts:
            conflict_type = ConflictType.DEPENDENCY_CONFLICT
            severity = "medium"
        elif structural_conflicts:
            conflict_type = ConflictType.STRUCTURAL_CONFLICT
            severity = "medium"
        else:
            conflict_type = ConflictType.CONCURRENT_MODIFICATION
            severity = "low"
        
        # Generate resolution suggestions
        suggestions = self._generate_resolution_suggestions(
            conflict_type, local_version, remote_version
        )
        
        return ConflictDescriptor(
            conflict_id=str(uuid.uuid4()),
            conflict_type=conflict_type,
            entity_type=local_version.entity_type,
            entity_id=local_version.entity_id,
            local_version=local_version,
            remote_version=remote_version,
            conflict_details={
                "structural_conflicts": structural_conflicts,
                "dependency_conflicts": dependency_conflicts,
                "semantic_conflicts": semantic_conflicts
            },
            resolution_suggestions=suggestions,
            severity=severity
        )
    
    def _detect_structural_conflicts(self, local_content: Dict, remote_content: Dict) -> List[Dict[str, Any]]:
        """Detect structural differences in entity content."""
        
        conflicts = []
        
        # Check for field-level conflicts
        all_fields = set(local_content.keys()) | set(remote_content.keys())
        
        for field in all_fields:
            local_value = local_content.get(field)
            remote_value = remote_content.get(field)
            
            if local_value != remote_value:
                conflicts.append({
                    "field": field,
                    "local_value": local_value,
                    "remote_value": remote_value,
                    "conflict_type": "field_value_mismatch"
                })
        
        return conflicts
    
    def _generate_resolution_suggestions(self, conflict_type: ConflictType,
                                       local_version: EntityVersion, 
                                       remote_version: EntityVersion) -> List[Dict[str, Any]]:
        """Generate resolution suggestions based on conflict analysis."""
        
        suggestions = []
        
        if conflict_type == ConflictType.CONCURRENT_MODIFICATION:
            suggestions.extend([
                {
                    "type": "take_local",
                    "description": "Keep local changes",
                    "confidence": 0.5
                },
                {
                    "type": "take_remote", 
                    "description": "Accept remote changes",
                    "confidence": 0.5
                },
                {
                    "type": "merge_fields",
                    "description": "Merge non-conflicting fields automatically",
                    "confidence": 0.8
                }
            ])
        elif conflict_type == ConflictType.STRUCTURAL_CONFLICT:
            suggestions.extend([
                {
                    "type": "manual_merge",
                    "description": "Manual resolution required due to structural changes",
                    "confidence": 0.3
                },
                {
                    "type": "create_variant",
                    "description": "Create separate entity variants",
                    "confidence": 0.6
                }
            ])
        
        return suggestions
```

```python
class ConflictResolutionEngine:
    """Handles automatic and guided manual conflict resolution."""
    
    def __init__(self, conflict_detector, version_manager):
        self.conflict_detector = conflict_detector
        self.version_manager = version_manager
    
    def resolve_conflicts_auto(self, conflicts: List[ConflictDescriptor]) -> Dict[str, Any]:
        """Attempt automatic resolution of conflicts."""
        
        auto_resolved = []
        manual_required = []
        
        for conflict in conflicts:
            if conflict.severity == "low":
                resolution = self._auto_resolve_conflict(conflict)
                if resolution:
                    auto_resolved.append(resolution)
                else:
                    manual_required.append(conflict)
            else:
                manual_required.append(conflict)
        
        return {
            "auto_resolved": auto_resolved,
            "manual_required": manual_required,
            "success_rate": len(auto_resolved) / len(conflicts) if conflicts else 1.0
        }
    
    def resolve_conflict_manual(self, conflict_id: str, resolution_choice: Dict[str, Any],
                               resolved_by: str) -> Dict[str, Any]:
        """Apply manual conflict resolution."""
        
        # Get conflict details
        conflict = self._get_conflict(conflict_id)
        if not conflict:
            raise ValueError(f"Conflict {conflict_id} not found")
        
        # Apply resolution based on choice
        if resolution_choice["type"] == "take_local":
            merged_content = conflict.local_version.content
        elif resolution_choice["type"] == "take_remote":
            merged_content = conflict.remote_version.content
        elif resolution_choice["type"] == "custom_merge":
            merged_content = resolution_choice["merged_content"]
        else:
            raise ValueError(f"Unknown resolution type: {resolution_choice['type']}")
        
        # Create resolved version
        resolved_version = self.version_manager.create_version(
            entity_type=conflict.entity_type,
            entity_id=conflict.entity_id,
            content=merged_content,
            author_id=resolved_by,
            state=ChangeState.MERGED
        )
        
        # Record resolution
        self._record_conflict_resolution(conflict_id, resolution_choice, resolved_version.id)
        
        return {
            "status": "resolved",
            "conflict_id": conflict_id,
            "resolved_version_id": resolved_version.id,
            "resolution_type": resolution_choice["type"]
        }
    
    def _auto_resolve_conflict(self, conflict: ConflictDescriptor) -> Optional[Dict[str, Any]]:
        """Attempt automatic resolution using CRDT and heuristics."""
        
        # Use highest confidence suggestion
        best_suggestion = max(conflict.resolution_suggestions, 
                            key=lambda x: x.get("confidence", 0),
                            default=None)
        
        if not best_suggestion or best_suggestion["confidence"] < 0.7:
            return None  # Confidence too low for auto-resolution
        
        # Apply auto-resolution
        try:
            if best_suggestion["type"] == "merge_fields":
                merged_content = self._merge_fields_automatically(
                    conflict.local_version.content,
                    conflict.remote_version.content
                )
                
                resolved_version = self.version_manager.create_version(
                    entity_type=conflict.entity_type,
                    entity_id=conflict.entity_id,
                    content=merged_content,
                    author_id="system",
                    state=ChangeState.MERGED
                )
                
                return {
                    "conflict_id": conflict.conflict_id,
                    "resolution_type": "auto_merge_fields",
                    "resolved_version_id": resolved_version.id
                }
                
        except Exception as e:
            # Auto-resolution failed, require manual intervention
            return None
        
        return None
```

class ChangeAnalyticsEngine:
    """Provides analytical views and insights using DuckDB."""
    
    def __init__(self, duckdb_conn: duckdb.DuckDBPyConnection, s3_config: Dict[str, str]):
        self.duckdb_conn = duckdb_conn
        self.s3_config = s3_config
        self._setup_analytical_views()
    
    def _setup_analytical_views(self):
        """Create analytical views for common queries."""
        
        # Recent changes across all users
        self.duckdb_conn.execute(f"""
        CREATE OR REPLACE VIEW recent_changes AS
        SELECT *,
               date_diff('day', created_at::timestamp, now()) as days_ago
        FROM read_parquet('s3://{self.s3_config["bucket"]}/changes/*/*/*.parquet')
        WHERE created_at::timestamp > now() - INTERVAL '30 days'
        ORDER BY created_at DESC;
        """)
        
        # Change velocity by user
        self.duckdb_conn.execute(f"""
        CREATE OR REPLACE VIEW user_change_velocity AS
        SELECT author_id,
               date_trunc('day', created_at::timestamp) as change_date,
               count(*) as changes_count,
               count(distinct entity_id) as entities_modified,
               count(distinct batch_id) as batches_count
        FROM read_parquet('s3://{self.s3_config["bucket"]}/changes/*/*/*.parquet')
        WHERE created_at::timestamp > now() - INTERVAL '90 days'
        GROUP BY author_id, date_trunc('day', created_at::timestamp)
        ORDER BY change_date DESC, changes_count DESC;
        """)
        
        # Entity modification patterns
        self.duckdb_conn.execute(f"""
        CREATE OR REPLACE VIEW entity_modification_patterns AS
        SELECT entity_type,
               entity_id,
               count(*) as total_modifications,
               count(distinct author_id) as unique_authors,
               min(created_at::timestamp) as first_modified,
               max(created_at::timestamp) as last_modified,
               date_diff('day', min(created_at::timestamp), max(created_at::timestamp)) as lifespan_days
        FROM read_parquet('s3://{self.s3_config["bucket"]}/changes/*/*/*.parquet')
        GROUP BY entity_type, entity_id
        ORDER BY total_modifications DESC;
        """)
        
        # Collaboration patterns
        self.duckdb_conn.execute(f"""
        CREATE OR REPLACE VIEW collaboration_patterns AS
        SELECT p.changeset_id,
               p.created_by as proposal_author,
               v.user_id as voter,
               v.vote,
               date_diff('hour', p.created_at::timestamp, v.voted_at::timestamp) as response_time_hours
        FROM read_parquet('s3://{self.s3_config["bucket"]}/proposals/*/metadata.parquet') p
        JOIN read_parquet('s3://{self.s3_config["bucket"]}/proposals/*/votes/*.parquet') v
          ON p.id = v.proposal_id
        WHERE p.created_at::timestamp > now() - INTERVAL '60 days';
        """)
    
    def get_change_summary(self, days: int = 30) -> Dict[str, Any]:
        """Get comprehensive change summary for specified period."""
        
        query = f"""
        SELECT 
            count(*) as total_changes,
            count(distinct entity_id) as entities_modified,
            count(distinct author_id) as active_users,
            count(distinct batch_id) as sync_operations,
            min(created_at) as period_start,
            max(created_at) as period_end
        FROM recent_changes
        WHERE days_ago <= {days};
        """
        
        result = self.duckdb_conn.execute(query).df()
        return result.iloc[0].to_dict() if not result.empty else {}
    
    def get_user_activity_report(self, user_id: Optional[str] = None, days: int = 30) -> pd.DataFrame:
        """Generate user activity report."""
        
        where_clause = f"AND author_id = '{user_id}'" if user_id else ""
        
        query = f"""
        SELECT author_id,
               sum(changes_count) as total_changes,
               sum(entities_modified) as total_entities,
               count(distinct change_date) as active_days,
               avg(changes_count) as avg_changes_per_day,
               max(changes_count) as max_changes_per_day
        FROM user_change_velocity
        WHERE date_diff('day', change_date, now()) <= {days} {where_clause}
        GROUP BY author_id
        ORDER BY total_changes DESC;
        """
        
        return self.duckdb_conn.execute(query).df()
    
    def get_entity_hotspots(self, limit: int = 50) -> pd.DataFrame:
        """Identify entities with highest modification frequency."""
        
        query = f"""
        SELECT entity_type, entity_id, total_modifications, unique_authors,
               lifespan_days, 
               total_modifications::float / greatest(lifespan_days, 1) as modification_rate
        FROM entity_modification_patterns
        WHERE total_modifications > 1
        ORDER BY modification_rate DESC, total_modifications DESC
        LIMIT {limit};
        """
        
        return self.duckdb_conn.execute(query).df()
    
    def get_collaboration_metrics(self, days: int = 60) -> Dict[str, Any]:
        """Analyze collaboration effectiveness."""
        
        query = f"""
        SELECT 
            count(distinct proposal_author) as proposal_authors,
            count(distinct voter) as voters,
            count(*) as total_votes,
            avg(response_time_hours) as avg_response_time_hours,
            sum(case when vote = 'approve' then 1 else 0 end)::float / count(*) as approval_rate
        FROM collaboration_patterns
        WHERE response_time_hours >= 0 AND response_time_hours <= 24 * {days};
        """
        
        result = self.duckdb_conn.execute(query).df()
        return result.iloc[0].to_dict() if not result.empty else {}
    
    def generate_change_impact_analysis(self, changeset_id: str) -> Dict[str, Any]:
        """Analyze the impact of a specific changeset."""
        
        query = f"""
        SELECT entity_type,
               count(*) as entities_affected,
               array_agg(distinct entity_id) as entity_ids
        FROM read_parquet('s3://{self.s3_config["bucket"]}/changes/*/*/*.parquet')
        WHERE changeset_id = '{changeset_id}'
        GROUP BY entity_type;
        """
        
        result = self.duckdb_conn.execute(query).df()
        
        return {
            "changeset_id": changeset_id,
            "total_entities": result['entities_affected'].sum() if not result.empty else 0,
            "impact_by_type": result.to_dict('records') if not result.empty else []
        }

#### 2.4.1 Incremental Sync Engine
```python
class IncrementalSyncEngine:
    """Handles efficient incremental synchronization using partitioned queries."""
    
    def __init__(self, duckdb_conn, s3_config, sqlite_conn):
        self.duckdb_conn = duckdb_conn
        self.s3_config = s3_config
        self.sqlite_conn = sqlite_conn
        
    def sync_incremental(self, since: datetime, until: Optional[datetime] = None,
                        entity_types: Optional[List[str]] = None) -> Dict[str, Any]:
        """Perform incremental sync for specified time range and entity types."""
        
        if until is None:
            until = datetime.now()
        
        # Generate partitioned query based on time range
        partitions = self._generate_time_partitions(since, until)
        
        sync_results = {
            "synced_changes": 0,
            "new_entities": 0,
            "updated_entities": 0,
            "sync_duration_seconds": 0,
            "partitions_processed": len(partitions)
        }
        
        start_time = time.time()
        
        for partition in partitions:
            partition_result = self._sync_partition(partition, entity_types)
            
            sync_results["synced_changes"] += partition_result["changes_count"]
            sync_results["new_entities"] += partition_result["new_entities"]
            sync_results["updated_entities"] += partition_result["updated_entities"]
        
        sync_results["sync_duration_seconds"] = time.time() - start_time
        
        # Update local sync metadata
        self._update_sync_metadata(since, until, sync_results)
        
        return sync_results
    
    def _generate_time_partitions(self, since: datetime, until: datetime) -> List[Dict[str, Any]]:
        """Generate time-based partition queries for efficient S3 access."""
        
        partitions = []
        current = since.date()
        end_date = until.date()
        
        while current <= end_date:
            partitions.append({
                "year": current.year,
                "month": current.month,
                "day": current.day,
                "s3_path_pattern": f"s3://{self.s3_config['bucket']}/changes/year={current.year}/month={current.month:02d}/day={current.day:02d}/*.parquet"
            })
            current += timedelta(days=1)
        
        return partitions
    
    def _sync_partition(self, partition: Dict[str, Any], 
                       entity_types: Optional[List[str]]) -> Dict[str, Any]:
        """Sync specific time partition."""
        
        # Build query with optional entity type filtering
        entity_filter = ""
        if entity_types:
            entity_list = "', '".join(entity_types)
            entity_filter = f"AND entity_type IN ('{entity_list}')"
        
        query = f"""
        SELECT * FROM read_parquet('{partition["s3_path_pattern"]}')
        WHERE 1=1 {entity_filter}
        ORDER BY created_at;
        """
        
        try:
            # Execute partitioned query
            result_df = self.duckdb_conn.execute(query).df()
            
            if result_df.empty:
                return {"changes_count": 0, "new_entities": 0, "updated_entities": 0}
            
            # Apply changes to local database
            return self._apply_changes_locally(result_df)
            
        except Exception as e:
            # Partition might not exist (no changes on that day)
            if "No files found" in str(e):
                return {"changes_count": 0, "new_entities": 0, "updated_entities": 0}
            raise
    
    def _apply_changes_locally(self, changes_df: pd.DataFrame) -> Dict[str, Any]:
        """Apply remote changes to local SQLite database."""
        
        new_entities = 0
        updated_entities = 0
        
        for _, change_row in changes_df.iterrows():
            # Check if entity exists locally
            entity_exists = self._entity_exists_locally(
                change_row['entity_type'], 
                change_row['entity_id']
            )
            
            # Create or update entity version
            version = self.version_manager.create_version(
                entity_type=change_row['entity_type'],
                entity_id=change_row['entity_id'],
                content=json.loads(change_row['content']),
                author_id=change_row['author_id'],
                state=ChangeState.MERGED  # Remote changes are already approved
            )
            
            if entity_exists:
                updated_entities += 1
            else:
                new_entities += 1
        
        return {
            "changes_count": len(changes_df),
            "new_entities": new_entities,
            "updated_entities": updated_entities
        }
    
    def get_optimal_sync_strategy(self, target_entity_count: int, 
                                 time_range_days: int) -> Dict[str, Any]:
        """Determine optimal sync strategy based on data volume."""
        
        # Estimate data volume
        estimated_changes = self._estimate_change_volume(time_range_days)
        
        if estimated_changes < 1000:
            strategy = "full_sync"
        elif target_entity_count < 10000:
            strategy = "incremental_daily"
        else:
            strategy = "incremental_hourly"
        
        return {
            "recommended_strategy": strategy,
            "estimated_changes": estimated_changes,
            "batch_size": min(1000, estimated_changes // 10) if estimated_changes > 0 else 100,
            "parallelization": estimated_changes > 5000
        }
```
