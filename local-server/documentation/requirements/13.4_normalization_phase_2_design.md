# 13.4 Normalization Phase 2 Design

## Overview

This document provides a detailed design for normalizing event handling and change tracking as specified in the requirements document 13.3. The changes focus on renaming and restructuring the event system to support a broader range of record types while maintaining backwards compatibility through proper migration.

## Architecture Changes

### 1. Model and Table Renaming

```python
class RecordTypeColumn(TypeDecorator):
    """
    Custom SQLAlchemy type for RecordType enum handling.
    
    This type decorator properly handles the conversion between
    Python RecordType enum instances and database string values,
    resolving SQLAlchemy enum conversion issues that occur with
    certain versions and configurations.
    
    Usage:
        record_type = Column(RecordTypeColumn(), nullable=False)
    """
    
    impl = String
    cache_ok = True
    
    def process_bind_param(self, value, dialect):
        """
        Convert Python enum to database string value.
        
        Args:
            value: RecordType enum instance or string
            dialect: SQLAlchemy dialect
            
        Returns:
            String value for database storage
        """
        if isinstance(value, RecordType):
            return value.value
        return value
    
    def process_result_value(self, value, dialect):
        """
        Convert database string value to Python enum.
        
        Args:
            value: String value from database
            dialect: SQLAlchemy dialect
            
        Returns:
            RecordType enum instance or None
        """
        if value is not None:
            return RecordType(value)
        return value
    
    def __repr__(self):
        return "RecordTypeColumn()"
```

```python
def _migrate_graph_events_to_change_events(self, connection: Connection) -> None:
    """Migrate graph_events records to change_events table with schema transformation."""
    logger.info("Migrating graph_events to change_events...")
    
    connection.execute(text("""
        INSERT INTO change_events (
            event_type, record_type, record_id, old_data, new_data, timestamp, processed
        )
        SELECT 
            event_type,
            CASE 
                WHEN entity_type IN ('layer', 'domain', 'term') THEN 'structure_node'
                WHEN entity_type = 'term_relationship' THEN 'structure_node_link'
                ELSE entity_type 
            END as record_type,
            -- Extract record_id from JSON data where possible
            CASE 
                WHEN new_data IS NOT NULL THEN json_extract(new_data, '$.id')
                WHEN old_data IS NOT NULL THEN json_extract(old_data, '$.id')
                ELSE NULL
            END as record_id,
            old_data, 
            new_data, 
            timestamp, 
            processed
        FROM graph_events
    """))
```

#### 5.2 Rollback Strategy

**Rollback Implementation:**
```python
def down(self, connection: Connection) -> None:
    """Rollback the migration."""
    logger.info("Rolling back Normalization Phase 2...")
    
    connection.execute(text("PRAGMA foreign_keys=off;"))
    
    # Recreate structure_node_events table
    self._recreate_structure_node_events_table(connection)
    
    # Migrate data back from change_events
    connection.execute(text("""
        INSERT INTO structure_node_events (
            event_type, node_type, node_id, old_data, new_data, timestamp, processed
        )
        SELECT 
            event_type,
            CASE record_type
                WHEN 'structure_node' THEN 
                    CASE 
                        WHEN json_extract(new_data, '$.node_type') IS NOT NULL THEN json_extract(new_data, '$.node_type')
                        WHEN json_extract(old_data, '$.node_type') IS NOT NULL THEN json_extract(old_data, '$.node_type')
                        ELSE 'term'  -- fallback
                    END
                ELSE record_type
            END as node_type,
            record_id as node_id,
            old_data,
            new_data,
            timestamp,
            processed
        FROM change_events
        WHERE record_type IN ('structure_node', 'structure_node_link')
    """))
    
    # Drop change_events table
    connection.execute(text("DROP TABLE IF EXISTS change_events;"))
    
    connection.execute(text("PRAGMA foreign_keys=on;"))
```

```python
def _validate_normalization_migration(self, connection: Connection) -> None:
    """Validate normalization phase 2 migration."""
    
    # Verify table rename
    change_events_exists = connection.execute(text("""
        SELECT name FROM sqlite_master WHERE type='table' AND name='change_events'
    """)).fetchone()
    
    if not change_events_exists:
        raise Exception("change_events table was not created")
    
    # Verify column renames
    columns = connection.execute(text("PRAGMA table_info(change_events)")).fetchall()
    column_names = [col[1] for col in columns]
    
    required_columns = ['record_type', 'record_id']
    missing_columns = [col for col in required_columns if col not in column_names]
    if missing_columns:
        raise Exception(f"Missing columns in change_events: {missing_columns}")
    
    # Verify predicate triggers exist
    triggers = connection.execute(text("""
        SELECT name FROM sqlite_master WHERE type='trigger' AND name LIKE '%predicate%'
    """)).fetchall()
    
    if len(triggers) < 3:  # insert, update, delete
        raise Exception("Predicate triggers were not created")
```
