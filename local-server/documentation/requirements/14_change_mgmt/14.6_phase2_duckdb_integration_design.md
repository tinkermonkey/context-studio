import duckdb
from typing import Dict, Any, Optional

class DuckDBConfig:
    def __init__(self, s3_config: Dict[str, str]):
        self.s3_config = s3_config
        self.connection = None
        
    def initialize_connection(self) -> duckdb.DuckDBPyConnection:
        """Initialize DuckDB connection with S3 configuration."""
        conn = duckdb.connect(':memory:')
        
        # Install and load required extensions
        conn.execute("INSTALL httpfs;")
        conn.execute("LOAD httpfs;")
        
        # Configure S3 access
        conn.execute(f"SET s3_region='{self.s3_config['region']}';")
        conn.execute(f"SET s3_access_key_id='{self.s3_config['access_key']}';")
        conn.execute(f"SET s3_secret_access_key='{self.s3_config['secret_key']}';")
        
        if 'endpoint' in self.s3_config:
            conn.execute(f"SET s3_endpoint='{self.s3_config['endpoint']}';")
            
        # Test connection
        self._test_s3_connection(conn)
        
        self.connection = conn
        return conn
    
    def _test_s3_connection(self, conn: duckdb.DuckDBPyConnection):
        """Test S3 connectivity."""
        try:
            # Try to list objects in bucket
            test_query = f"SELECT * FROM read_parquet('s3://{self.s3_config['bucket']}/test.parquet') LIMIT 1;"
            conn.execute(test_query)
        except Exception:
            # Expected if test file doesn't exist
            pass

class S3StorageSchema:
    """Defines the S3 storage structure and partitioning strategy."""
    
    @staticmethod
    def get_changes_path(bucket: str, year: int, month: int, day: int, 
                        batch_id: str, user_id: str) -> str:
        """Generate S3 path for change batch."""
        return f"s3://{bucket}/changes/year={year}/month={month:02d}/day={day:02d}/batch_{batch_id}_{user_id}.parquet"
    
    @staticmethod
    def get_metadata_path(bucket: str, entity_type: str) -> str:
        """Generate S3 path for entity metadata."""
        return f"s3://{bucket}/metadata/{entity_type}/metadata.parquet"
    
    @staticmethod
    def get_checkpoint_path(bucket: str, checkpoint_date: str) -> str:
        """Generate S3 path for checkpoint."""
        return f"s3://{bucket}/checkpoints/checkpoint_{checkpoint_date}.parquet"

### 2.2 Change Extraction and Serialization

#### 2.2.1 Change Extractor
```python
from datetime import datetime, date
import pandas as pd
import uuid
from dataclasses import dataclass, asdict
from typing import List, Dict, Any, Optional

@dataclass
class ChangeRecord:
    """Represents a single change record for S3 storage."""
    change_id: str
    entity_type: str
    entity_id: str
    version_id: str
    version_number: int
    operation: str  # CREATE, UPDATE, DELETE
    content: Dict[str, Any]
    author_id: str
    created_at: str  # ISO timestamp
    batch_id: str
    changeset_id: Optional[str] = None
    metadata: Optional[Dict[str, Any]] = None

class ChangeExtractor:
    """Extracts changes from SQLite and prepares for S3 sync."""
    
    def __init__(self, sqlite_conn, version_manager):
        self.sqlite_conn = sqlite_conn
        self.version_manager = version_manager
        
    def extract_pending_changes(self, since: Optional[datetime] = None) -> List[ChangeRecord]:
        """Extract changes that need to be synchronized to S3."""
        
        query = """
        SELECT ev.id, ev.entity_type, ev.entity_id, ev.version_number,
               ev.content, ev.author_id, ev.created_at, ev.state,
               ce.operation, ce.id as change_event_id
        FROM entity_versions ev
        JOIN change_events ce ON ev.id = ce.version_id
        WHERE ev.state IN ('WORKING', 'STAGED', 'PROPOSED')
        """
        
        if since:
            query += f" AND ev.created_at > '{since.isoformat()}'"
            
        query += " ORDER BY ev.created_at"
        
        cursor = self.sqlite_conn.execute(query)
        rows = cursor.fetchall()
        
        batch_id = str(uuid.uuid4())
        changes = []
        
        for row in rows:
            change = ChangeRecord(
                change_id=row['change_event_id'],
                entity_type=row['entity_type'],
                entity_id=row['entity_id'],
                version_id=row['id'],
                version_number=row['version_number'],
                operation=row['operation'],
                content=json.loads(row['content']),
                author_id=row['author_id'],
                created_at=row['created_at'],
                batch_id=batch_id
            )
            changes.append(change)
            
        return changes
    
    def create_change_dataframe(self, changes: List[ChangeRecord]) -> pd.DataFrame:
        """Convert change records to pandas DataFrame for Parquet serialization."""
        
        if not changes:
            return pd.DataFrame()
            
        # Convert to dictionaries
        change_dicts = [asdict(change) for change in changes]
        
        # Create DataFrame
        df = pd.DataFrame(change_dicts)
        
        # Optimize data types
        df['created_at'] = pd.to_datetime(df['created_at'])
        df['version_number'] = df['version_number'].astype('int32')
        
        # Convert complex objects to JSON strings for Parquet
        df['content'] = df['content'].apply(json.dumps)
        df['metadata'] = df['metadata'].apply(lambda x: json.dumps(x) if x else None)
        
        return df
```

#### 2.2.2 Parquet Writer
```python
class ParquetWriter:
    """Handles writing DataFrames to S3 as Parquet files."""
    
    def __init__(self, duckdb_conn: duckdb.DuckDBPyConnection):
        self.duckdb_conn = duckdb_conn
        
    def write_changes_batch(self, df: pd.DataFrame, s3_path: str) -> bool:
        """Write change batch to S3 as Parquet."""
        
        if df.empty:
            return True
            
        try:
            # Register DataFrame as temporary table
            self.duckdb_conn.register('temp_changes', df)
            
            # Write to S3 with compression
            query = f"""
            COPY (SELECT * FROM temp_changes) 
            TO '{s3_path}' 
            (FORMAT 'parquet', COMPRESSION 'zstd')
            """
            
            self.duckdb_conn.execute(query)
            
            # Clean up
            self.duckdb_conn.unregister('temp_changes')
            
            return True
            
        except Exception as e:
            print(f"Error writing to S3: {e}")
            return False
            
    def write_metadata(self, metadata: Dict[str, Any], s3_path: str) -> bool:
        """Write metadata to S3."""
        
        try:
            # Convert metadata to DataFrame
            df = pd.DataFrame([metadata])
            
            # Register and write
            self.duckdb_conn.register('temp_metadata', df)
            
            query = f"""
            COPY (SELECT * FROM temp_metadata)
            TO '{s3_path}'
            (FORMAT 'parquet', COMPRESSION 'zstd')
            """
            
            self.duckdb_conn.execute(query)
            self.duckdb_conn.unregister('temp_metadata')
            
            return True
            
        except Exception as e:
            print(f"Error writing metadata: {e}")
            return False
```

### 2.3 S3 Operations Manager

#### 2.3.1 S3 Sync Manager
```python
class S3SyncManager:
    """Manages synchronization operations with S3."""
    
    def __init__(self, duckdb_conn: duckdb.DuckDBPyConnection, 
                 s3_config: Dict[str, str], sqlite_conn):
        self.duckdb_conn = duckdb_conn
        self.s3_config = s3_config
        self.sqlite_conn = sqlite_conn
        self.change_extractor = ChangeExtractor(sqlite_conn, None)
        self.parquet_writer = ParquetWriter(duckdb_conn)
        
    def push_changes(self, author_id: str) -> Dict[str, Any]:
        """Push local changes to S3."""
        
        # Extract pending changes
        changes = self.change_extractor.extract_pending_changes()
        
        if not changes:
            return {"status": "success", "message": "No changes to push", "changes_count": 0}
        
        # Group changes by date for partitioning
        changes_by_date = {}
        for change in changes:
            change_date = datetime.fromisoformat(change.created_at).date()
            if change_date not in changes_by_date:
                changes_by_date[change_date] = []
            changes_by_date[change_date].append(change)
        
        # Push each date partition
        pushed_batches = []
        for change_date, date_changes in changes_by_date.items():
            
            # Create DataFrame
            df = self.change_extractor.create_change_dataframe(date_changes)
            
            # Generate S3 path
            batch_id = str(uuid.uuid4())
            s3_path = S3StorageSchema.get_changes_path(
                self.s3_config['bucket'],
                change_date.year,
                change_date.month,
                change_date.day,
                batch_id,
                author_id
            )
            
            # Write to S3
            if self.parquet_writer.write_changes_batch(df, s3_path):
                pushed_batches.append({
                    "date": change_date.isoformat(),
                    "path": s3_path,
                    "changes_count": len(date_changes)
                })
                
                # Mark changes as pushed (update state)
                self._mark_changes_pushed(date_changes)
            else:
                return {"status": "error", "message": f"Failed to push changes for {change_date}"}
        
        return {
            "status": "success", 
            "message": "Changes pushed successfully",
            "batches": pushed_batches,
            "total_changes": len(changes)
        }
    
    def pull_changes(self, since: Optional[datetime] = None) -> Dict[str, Any]:
        """Pull remote changes from S3."""
        
        try:
            # Build query for remote changes
            where_clause = ""
            if since:
                where_clause = f"WHERE created_at > '{since.isoformat()}'"
            
            query = f"""
            SELECT * FROM read_parquet('s3://{self.s3_config["bucket"]}/changes/*/*/*.parquet')
            {where_clause}
            ORDER BY created_at
            """
            
            # Execute query
            result = self.duckdb_conn.execute(query).df()
            
            if result.empty:
                return {"status": "success", "message": "No new changes", "changes_count": 0}
            
            # Convert to change records
            changes = self._dataframe_to_changes(result)
            
            return {
                "status": "success",
                "message": f"Retrieved {len(changes)} changes",
                "changes_count": len(changes),
                "changes": changes
            }
            
        except Exception as e:
            return {"status": "error", "message": f"Failed to pull changes: {str(e)}"}
    
    def get_sync_status(self) -> Dict[str, Any]:
        """Get current synchronization status."""
        
        # Count pending local changes
        pending_changes = self.change_extractor.extract_pending_changes()
        
        # Get last sync timestamps
        last_push = self._get_last_push_timestamp()
        last_pull = self._get_last_pull_timestamp()
        
        return {
            "pending_push_count": len(pending_changes),
            "last_push": last_push.isoformat() if last_push else None,
            "last_pull": last_pull.isoformat() if last_pull else None,
            "s3_connection": self._test_s3_connection(),
            "local_changes": len(pending_changes) > 0
        }
    
    def _mark_changes_pushed(self, changes: List[ChangeRecord]):
        """Mark changes as pushed in local database."""
        
        change_ids = [change.change_id for change in changes]
        placeholders = ','.join(['?' for _ in change_ids])
        
        query = f"""
        UPDATE change_events 
        SET metadata = json_set(
            COALESCE(metadata, '{{}}'), 
            '$.pushed_at', 
            datetime('now')
        )
        WHERE id IN ({placeholders})
        """
        
        self.sqlite_conn.execute(query, change_ids)
        self.sqlite_conn.commit()
    
    def _dataframe_to_changes(self, df: pd.DataFrame) -> List[Dict[str, Any]]:
        """Convert DataFrame back to change records."""
        
        changes = []
        for _, row in df.iterrows():
            change = {
                "change_id": row['change_id'],
                "entity_type": row['entity_type'],
                "entity_id": row['entity_id'],
                "version_id": row['version_id'],
                "operation": row['operation'],
                "content": json.loads(row['content']),
                "author_id": row['author_id'],
                "created_at": row['created_at'].isoformat(),
                "batch_id": row['batch_id']
            }
            changes.append(change)
            
        return changes
    
    def _get_last_push_timestamp(self) -> Optional[datetime]:
        """Get timestamp of last successful push."""
        
        query = """
        SELECT MAX(json_extract(metadata, '$.pushed_at')) as last_push
        FROM change_events 
        WHERE json_extract(metadata, '$.pushed_at') IS NOT NULL
        """
        
        result = self.sqlite_conn.execute(query).fetchone()
        if result and result['last_push']:
            return datetime.fromisoformat(result['last_push'])
        return None
    
    def _get_last_pull_timestamp(self) -> Optional[datetime]:
        """Get timestamp of last successful pull."""
        
        # Implementation would track pull timestamps in local metadata table
        # For now, return None
        return None
    
    def _test_s3_connection(self) -> bool:
        """Test S3 connection."""
        
        try:
            query = f"SELECT 1 FROM read_parquet('s3://{self.s3_config['bucket']}/test/*.parquet') LIMIT 1"
            self.duckdb_conn.execute(query)
            return True
        except Exception:
            return False
```
